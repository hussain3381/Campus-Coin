database Database Schema Design: Multi-Tenant + System Defaults (REQ-CAT-04) database/migrations/create_categories_table.php 
Schema::create('categories', function (Blueprint $table) {
    $table->id();
    // REQ-CAT-04: user_id is NULL for Admin System Defaults; populated for Student custom envelopes
    $table->foreignId('user_id')->nullable()->constrained()->onDelete('cascade');
    $table->string('name');
    $table->string('icon')->default('payments');
    $table->string('color')->default('#10b981');
    $table->decimal('monthly_limit', 10, 2)->nullable();
    $table->boolean('is_active')->default(true);
    $table->timestamps();
});
